Indexes:
    "channel_endpoints_pkey" PRIMARY KEY, btree (id)
    "channel_endpoints_kind_identifier_uidx" UNIQUE, btree (channel_kind, identifier)
    "channel_endpoints_public_stream_url_idx" btree (public_stream_url)
    "channel_endpoints_public_webhook_url_idx" btree (public_webhook_url)
    "channel_endpoints_workspace_kind_agent_idx" btree (workspace_id, channel_kind, attached_agent_id)
Check constraints:
    "channel_endpoints_attachment_check" CHECK (attached_agent_id IS NOT NULL OR routing_rules_id IS NOT NULL)
    "channel_endpoints_channel_kind_check" CHECK (channel_kind = ANY (ARRAY['voice'::text, 'whatsapp'::text, 'messenger'::text, 'instagram'::text, 'web_chat'::text, 'sms'::text]))
Foreign-key constraints:
    "channel_endpoints_attached_agent_id_agents_id_fk" FOREIGN KEY (attached_agent_id) REFERENCES agents(id)
    "channel_endpoints_connection_id_channel_connections_id_fk" FOREIGN KEY (connection_id) REFERENCES channel_connections(id) ON DELETE CASCADE
    "channel_endpoints_workspace_id_organization_id_fk" FOREIGN KEY (workspace_id) REFERENCES organization(id) ON DELETE CASCADE
Referenced by:
    TABLE "conversations" CONSTRAINT "conversations_channel_endpoint_id_channel_endpoints_id_fk" FOREIGN KEY (channel_endpoint_id) REFERENCES channel_endpoints(id)
    TABLE "messaging_threads" CONSTRAINT "messaging_threads_channel_endpoint_id_channel_endpoints_id_fk" FOREIGN KEY (channel_endpoint_id) REFERENCES channel_endpoints(id)
    TABLE "routing_rules" CONSTRAINT "routing_rules_channel_endpoint_id_channel_endpoints_id_fk" FOREIGN KEY (channel_endpoint_id) REFERENCES channel_endpoints(id) ON DELETE CASCADE
Triggers:
    channel_endpoint_kind_check BEFORE INSERT OR UPDATE ON channel_endpoints FOR EACH ROW EXECUTE FUNCTION channel_endpoint_kind_matches()
Access method: heap
